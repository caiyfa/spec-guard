```python
"""sg_core.db_access — 只读 DB 访问统一出口（DB 面板/db_evidence/试跑三端共享）。

设计 2026-08-25（docs/superpowers/specs/2026-08-25-DB证据面板与辅助会话-design.md §A1）：
- 守卫链 = ReadOnlyDb._guard（SELECT-only/危险子句/多语句）∪ attribution.SQL_DENY_RE
  （逃逸黑名单：LOCK IN SHARE MODE/LOCK TABLES/HANDLER/LOAD DATA/DO SLEEP 等）——
  任何一端单独的守卫都不删减，统一为并集。
- 进程级连接缓存：键 = (workspace, system, env)；ReadOnlyDb._ensure_alive 自带
  ping 重连，实例可长期复用。TTL 兜底 300s 防配置漂移。
- 插值：vars=None 跳过（db_evidence 现状语义）；vars 传入时走 engine.interpolation，
  缺键诚实报变量名（编辑态试跑引用运行时变量的引导依据）。
"""
from __future__ import annotations

import re
import threading
import time

SG_ROOT = __import__("pathlib").Path(__file__).resolve().parent.parent

# 并集守卫（attribution.py SQL_DENY_RE 全量搬运 + ReadOnlyDb._FORBIDDEN_SELECT 语义经 _guard 覆盖）
_DENY_RE = re.compile(
    r"\b(FOR\s+UPDATE|LOCK\s+IN\s+SHARE\s+MODE|INTO\s+(OUT|DUMP)FILE|LOCK\s+TABLES|"
    r"HANDLER|LOAD\s+DATA|LOAD\s+XML|DO\s+SLEEP)\b",
    re.IGNORECASE)
_ALLOWED_RE = re.compile(r"^\s*(SELECT|SHOW|DESCRIBE|DESC|EXPLAIN)\b", re.IGNORECASE)


class DbAccessError(Exception):
    """db_access 统一错误（守卫拒绝/未配置/插值失败/连接失败）。"""

    def __init__(self, message: str, hint: str = ""):
        super().__init__(message)
        self.hint = hint


def _guard_sql(sql: str) -> None:
    """只读守卫（连接建立前触发；文案与 ReadOnlyDb._guard 同风格）。"""
    s = sql or ""
    if not _ALLOWED_RE.match(s):
        raise DbAccessError(f"只读守卫拒绝非查询语句: {s[:80]}")
    m = _DENY_RE.search(s)
    if m:
        raise DbAccessError(f"只读守卫拒绝危险子句: {m.group(1)}")
    body = s.strip().rstrip(";")
    if ";" in body:
        raise DbAccessError(f"只读守卫拒绝多语句 SQL: {s[:80]}")


def _interpolate(sql: str, vars: dict | None) -> str:
    """vars=None 原样返回；否则插值，缺键报变量名。"""
    if vars is None:
        return sql
    from engine.interpolation import render, RenderError
    try:
        out = render(sql, vars or {})
        return out if isinstance(out, str) else str(out)
    except RenderError as e:
        raise DbAccessError(f"SQL 插值失败: {e}")
    except KeyError as e:
        raise DbAccessError(f"SQL 插值失败: {e}")


def _resolve_system() -> str:
    import os
    from specstd.workspace import resolve_system
    return resolve_system() or os.environ.get("SPEC_GUARD_SYSTEM", "tdm")


def _load_env_cfg(env_name: str) -> dict:
    import importlib
    system = _resolve_system()
    return importlib.import_module(f"systems.{system}").load_env(env_name) or {}


# ── 连接缓存 ──
_CACHE: dict[tuple, tuple[dict, str, float]] = {}  # key -> (dbs, default_name, cached_at)
_CACHE_LOCK = threading.Lock()
_CACHE_TTL = 300.0


def _cache_key(env_name: str) -> tuple:
    try:
        from specstd.workspace import get_context_workspace
        ws = get_context_workspace() or ""
    except Exception:
        ws = ""
    return (ws, _resolve_system(), env_name)


def invalidate_cache() -> None:
    """清空连接缓存（配置/工作区切换场景兜底）。"""
    with _CACHE_LOCK:
        _CACHE.clear()


def get_db_connections(env_name: str = "") -> tuple[dict, str]:
    """取命名连接注册表（进程级缓存；DB 不可达的 schema 跳过并 WARN——与
    engine.db_oracle.build_db_connections 同语义，但保留实例复用）。

    返回 (dbs: dict[str, ReadOnlyDb], default_name: str)。
    """
    import os
    env_name = env_name or os.environ.get("SG_ENV", "sit")
    key = _cache_key(env_name)
    now = time.time()
    with _CACHE_LOCK:
        cached = _CACHE.get(key)
        if cached and (now - cached[2]) < _CACHE_TTL:
            return cached[0], cached[1]
    cfg = _load_env_cfg(env_name)
    db_cfg = cfg.get("db") or {}
    if not db_cfg:
        raise DbAccessError(
            f"环境配置未设置 db 段（env={env_name}）",
            hint="请在系统配置页补充 DB 只读连接，或检查 systems 配置包 env yaml")
    # 建连（每 schema 探活；失败跳过该 schema——诚实降级）
    from engine.db_oracle import ReadOnlyDb, normalize_db_config
    conns_cfg, default_name = normalize_db_config(db_cfg)
    dbs: dict = {}
    for name, c in conns_cfg.items():
        if not c.get("host") or not c.get("user"):
            continue
        try:
            db = ReadOnlyDb(
                host=c["host"], port=c.get("port", 3306), user=c["user"],
                password=c.get("password", ""), database=c.get("name", "hr_center"),
                charset=c.get("charset", "utf8mb4"))
            db.scalar("SELECT 1")
            dbs[name] = db
        except Exception as e:
            print(f"[SpecGuard WARN] DB 连接失败（schema={name}），该 schema 不可用: {e}")
    with _CACHE_LOCK:
        _CACHE[key] = (dbs, default_name, time.time())
    return dbs, default_name


def execute_readonly(sql: str, schema: str = "", max_rows: int = 1000,
                     vars: dict | None = None, env_name: str = "",
                     _bypass_cache: bool = False) -> dict:
    """统一只读执行：守卫 → 插值 → 执行。

    返回 {rows: list[dict], row_count: int, truncated: bool, elapsed_ms: int,
    sql_rendered: str}。守卫/配置/插值/连接错误统一抛 DbAccessError（hint 可选）。
    _bypass_cache 仅测试用（绕过连接缓存强制走装配路径）。
    """
    import time as _time
    _guard_sql(sql)
    sql_rendered = _interpolate(sql, vars)
    if _bypass_cache:
        invalidate_cache()
    try:
        dbs, default_name = get_db_connections(env_name)
    except DbAccessError:
        raise
    if not dbs:
        raise DbAccessError(
            "DB 连接不可用（所有 schema 均连接失败或未配置）",
            hint="请检查系统配置页的 DB 连接配置")
    db = dbs.get(schema) if schema else dbs.get(default_name)
    if db is None:
        raise DbAccessError(
            f"schema {schema or default_name!r} 不存在或不可用",
            hint=f"可用 schema: {', '.join(sorted(dbs))}")
    t0 = _time.time()
    try:
        rows = db.query(sql_rendered, max_rows=max_rows)
    except DbAccessError:
        raise
    except Exception as e:
        raise DbAccessError(f"DB 查询失败: {e}", hint="请检查 SQL 语法与数据源连通性")
    return {
        "rows": rows,
        "row_count": len(rows),
        "truncated": db.last_truncated,
        "elapsed_ms": int((_time.time() - t0) * 1000),
        "sql_rendered": sql_rendered,
    }
```

```python
from pydantic import BaseModel

from services.auth import get_current_user

router = APIRouter(prefix="/api/db", tags=["db-console"])


class QueryBody(BaseModel):
    sql: str
    schema: str = ""
    vars: dict | None = None


@router.get("/schemas")
def list_schemas(user=Depends(get_current_user)):
    """当前工作区 env 配置的 schema 清单（名称 + default 标记 + 探活态）。"""
    from sg_core import db_access
    try:
        dbs, default_name = db_access.get_db_connections()
        return {"schemas": [
            {"name": n, "is_default": n == default_name, "alive": True}
            for n in sorted(dbs)
        ], "default": default_name}
    except db_access.DbAccessError:
        return {"schemas": [], "default": ""}
    except Exception:
        return {"schemas": [], "default": ""}


@router.post("/query")
def query(body: QueryBody, user=Depends(get_current_user)):
    """只读查询（守卫→插值→执行）；守卫/配置/执行错误统一 400 + 明细。"""
    from sg_core import db_access

    t_start = time.time()
    try:
        out = db_access.execute_readonly(
            body.sql, schema=body.schema or "", vars=body.vars)
    except db_access.DbAccessError as e:
        detail = str(e)
        if e.hint:
            detail = f"{detail}（{e.hint}）"
        raise HTTPException(status_code=400, detail=detail)

    # 审计（失败静默不阻断——T-082 纪律）
    try:
        from sg_core import user_store as _us
        _us.audit(
            user.emp_no, "db_query", target=body.schema or "default",
            detail=json.dumps({
                "sql": body.sql[:500], "rows": out["row_count"],
                "elapsed_ms": out["elapsed_ms"],
            }, ensure_ascii=False),
            db_path=_us.DB_PATH)
    except Exception:
        pass

    return {
        "rows": out["rows"],
        "row_count": out["row_count"],
        "truncated": out["truncated"],
        "elapsed_ms": out["elapsed_ms"],
        "sql_rendered": out["sql_rendered"],
        "total_ms": int((time.time() - t_start) * 1000),
    }


import json  # noqa: E402  （audit detail 用）
```

```tsx
// workbench/frontend/src/components/SqlResultTable.tsx
/** 只读 SQL 结果表格（设计 2026-08-25 §A5）——三处复用：
 * AgentChat db_evidence 工具块（G3）/ DB 面板结果区（A3）/ SqlAuxPanel 试跑结果（C2）。 */
export interface SqlResultData {
  rows: Record<string, unknown>[];
  row_count?: number;
  truncated?: boolean;
}

export function SqlResultTable({ rows, truncated, row_count }: SqlResultData & { truncated?: boolean; row_count?: number }) {
  const count = row_count ?? rows.length;
  if (!rows.length) {
    return <div className="text-xs text-gray-500 py-2">{count} 行（无数据）</div>;
  }
  const columns = Object.keys(rows[0]);
  return (
    <div className="text-xs">
      <div className="flex items-center gap-2 mb-1 text-gray-500">
        <span>{count} 行</span>
        {truncated && <span className="text-yellow-500/80">（结果超上限，已截断）</span>}
      </div>
      <div className="overflow-x-auto max-h-72 border border-border rounded">
        <table className="min-w-full">
          <thead>
            <tr className="bg-surface-3 text-left">
              {columns.map(c => (
                <th key={c} className="px-2 py-1 font-mono text-gray-400 whitespace-nowrap">{c}</th>
              ))}
            </tr>
          </thead>
          <tbody>
            {rows.map((r, i) => (
              <tr key={i} className="border-t border-border/50">
                {columns.map(c => (
                  <td key={c} className="px-2 py-0.5 font-mono text-gray-300 truncate max-w-[240px]"
                      title={String(r[c] ?? '')}>
                    {String(r[c] ?? '')}
                  </td>
                ))}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </div>
  );
}

/** 从工具结果文本解析 db_evidence 输出（含 rows 数组的 JSON）——解析失败返回 null（调用方回退纯文本）。 */
export function parseDbEvidenceResult(text: string): SqlResultData | null {
  try {
    const obj = JSON.parse(text);
    if (obj && Array.isArray(obj.rows)) {
      return { rows: obj.rows, row_count: obj.row_count ?? obj.rows.length };
    }
  } catch { /* 回退纯文本 */ }
  return null;
}
```

```tsx
// workbench/frontend/src/pages/DbConsole.tsx
/** DB 证据面板（设计 2026-08-25 §A3）——只读 SQL 执行 / schema 切换 / 结果表格 / 查询历史。
 * 状态持久化（基线①）：schema 选择与编辑器草稿 localStorage 恢复；
 * 三态分离（基线③）：加载中 / 空态 / 错误分别呈现。 */
import { useEffect, useState } from 'react';
import { useSearchParams } from 'react-router-dom';
import Editor from '@monaco-editor/react';
import { dbApi, DbQueryResult, DbSchema } from '../api/db';
import { SqlResultTable } from '../components/SqlResultTable';

const LS_SCHEMA = 'sg_db_schema';
const LS_SQL = 'sg_db_sql_draft';
const LS_HISTORY = 'sg_db_history';

interface HistoryItem { sql: string; schema: string; ts: number }

export function DbConsole() {
  const [searchParams] = useSearchParams();
  const [schemas, setSchemas] = useState<DbSchema[]>([]);
  const [schemasLoaded, setSchemasLoaded] = useState(false);
  const [schema, setSchema] = useState(() => localStorage.getItem(LS_SCHEMA) || '');
  const [sql, setSql] = useState(() => localStorage.getItem(LS_SQL) || 'SELECT 1');
  const [result, setResult] = useState<DbQueryResult | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState('');
  const [history, setHistory] = useState<HistoryItem[]>(() => {
    try { return JSON.parse(localStorage.getItem(LS_HISTORY) || '[]'); } catch { return []; }
  });

  useEffect(() => {
    dbApi.schemas().then(r => {
      setSchemas(r.schemas || []);
      if (!schema && r.default) setSchema(r.default);
    }).catch(() => setSchemas([])).finally(() => setSchemasLoaded(true));
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // 联动入口：?sql=... 回填并自动执行（A4 入口②跳转目标）
  useEffect(() => {
    const incoming = searchParams.get('sql');
    if (incoming) {
      setSql(incoming);
      runQuery(incoming, schema);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [searchParams]);

  useEffect(() => { localStorage.setItem(LS_SQL, sql); }, [sql]);
  useEffect(() => { localStorage.setItem(LS_SCHEMA, schema); }, [schema]);

  const runQuery = async (sqlText = sql, schemaName = schema) => {
    if (!sqlText.trim() || loading) return;
    setLoading(true); setError(''); setResult(null);
    try {
      const r = await dbApi.query({ sql: sqlText, schema: schemaName || undefined });
      setResult(r);
      setHistory(prev => {
        const next = [{ sql: sqlText, schema: schemaName, ts: Date.now() },
          ...prev.filter(h => h.sql !== sqlText)].slice(0, 50);
        localStorage.setItem(LS_HISTORY, JSON.stringify(next));
        return next;
      });
    } catch (e: any) {
      setError(e.message || '查询失败');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="max-w-6xl mx-auto p-4 space-y-3">
      <div className="flex items-center gap-3">
        <h1 className="text-lg font-medium">🗄 DB 证据</h1>
        <span className="text-xs text-gray-500">只读查询（共享只读账号，无行级权限）——仅 SELECT/SHOW/DESCRIBE/EXPLAIN</span>
      </div>

      {/* schema 切换器 */}
      <div className="flex items-center gap-2 text-xs">
        <span className="text-gray-500">schema</span>
        {schemasLoaded ? (schemas.length ? schemas.map(s => (
          <button key={s.name}
            onClick={() => setSchema(s.name)}
            className={`px-2 py-0.5 rounded border ${schema === s.name
              ? 'border-info text-info' : 'border-border text-gray-400 hover:text-gray-200'} ${!s.alive ? 'opacity-40' : ''}`}
            title={s.is_default ? '默认连接' : ''}>
            {s.name}{s.is_default ? ' ·default' : ''}{!s.alive ? '（连接失败）' : ''}
          </button>
        )) : <span className="text-gray-600">环境配置未设置 db 段（去系统配置页补充）</span>
        ) : <span className="text-gray-600">加载中…</span>}
      </div>

      {/* SQL 编辑器 */}
      <div className="border border-border rounded h-56">
        <Editor
          defaultLanguage="sql"
          value={sql}
          onChange={(v) => setSql(v || '')}
          theme="vs-dark"
          options={{ minimap: { enabled: false }, fontSize: 12, scrollBeyondLastLine: false }}
          onMount={(editor) => {
            editor.addCommand(0 /* Ctrl+Enter 由 keybinding 处理，此处回退按钮触发 */, () => {});
          }}
        />
      </div>

      <div className="flex items-center gap-3">
        <button onClick={() => runQuery()} disabled={loading}
          title="执行 (Ctrl+Enter)"
          className="px-3 py-1 text-xs rounded bg-info/20 text-info border border-info/40 hover:bg-info/30 disabled:opacity-50">
          {loading ? '查询中…' : '▶ 执行'}
        </button>
        {/* Ctrl+Enter 快捷执行 */}
        <span className="text-[10px] text-gray-600">Ctrl+Enter 执行</span>
      </div>

      {/* 三态分离：错误 / 结果 / 空态 */}
      {error && <div className="text-xs text-red-400 bg-red-900/10 border border-red-700/20 rounded p-2">❌ {error}</div>}
      {result && (
        <div className="space-y-1">
          <div className="text-[10px] text-gray-500">{result.elapsed_ms}ms{result.truncated ? ' · 结果超上限已截断' : ''}</div>
          <SqlResultTable rows={result.rows} row_count={result.row_count} truncated={result.truncated} />
        </div>
      )}
      {!result && !error && !loading && (
        <div className="text-xs text-gray-600 py-4">输入 SQL 后执行；结果仅在本页展示并记录查询历史</div>
      )}

      {/* 查询历史 */}
      {history.length > 0 && (
        <div>
          <div className="text-xs text-gray-500 mb-1">查询历史（{history.length}）</div>
          <div className="space-y-0.5 max-h-40 overflow-y-auto">
            {history.map((h, i) => (
              <button key={i} onClick={() => { setSql(h.sql); setSchema(h.schema); }}
                className="block w-full text-left text-[11px] font-mono text-gray-500 hover:text-gray-300 truncate"
                title={h.sql}>
                [{h.schema}] {h.sql}
              </button>
            ))}
          </div>
        </div>
      )}
    </div>
  );
}

export default DbConsole;
```

```python
        """批量更新首行 meta 字段（辅助会话快照指纹等运行时元数据，设计 §B4）。

        返回是否有效（文件不存在/首行非 meta 返回 False；值未变化返回 True 不写）。
        """
        _validate_session_id(session_id)
        path = self.sessions_dir / f"{session_id}.jsonl"
        if not path.is_file():
            return False
        lines = path.read_text(encoding="utf-8").splitlines()
        if not lines:
            return False
        try:
            meta = json.loads(lines[0])
            if meta.get("_type") != "session_meta":
                return False
            changed = False
            for k, v in fields.items():
                if meta.get(k) != v:
                    meta[k] = v
                    changed = True
            if not changed:
                return True
            lines[0] = json.dumps(meta, ensure_ascii=False)
            path.write_text("\n".join(lines) + "\n", encoding="utf-8")
            return True
        except (json.JSONDecodeError, IndexError):
            return False
```

```python
@router.post("/sessions/spawn")
def spawn_session(body: dict, user: CurrentUser = Depends(get_current_user)):
    """辅助会话派生（设计 2026-08-25 §B3）：从主会话派生专职 purpose 的辅助会话。

    幂等：同父会话 + 同 purpose 存在未归档 aux 会话 → 直接返回（created=False）；
    父会话的 aux 已归档时再派生新的。父会话必须存在且属当前用户。
    """
    parent_sid = str(body.get("parent_session_id", "") or "").strip()
    purpose = str(body.get("purpose", "") or "").strip()
    workspace = str(body.get("workspace", "") or "").strip()
    if not parent_sid or not purpose:
        raise HTTPException(status_code=400, detail="parent_session_id 与 purpose 必填")
    try:
        _validate_session_id(parent_sid)
    except ValueError:
        raise HTTPException(status_code=400, detail="非法 parent_session_id")
    pdef = load_purpose(purpose)
    if pdef is None or not (set(user.roles) & set(pdef.roles)):
        raise HTTPException(status_code=403, detail=f"用途不存在或无权限: {purpose}")
    try:
        sm = SessionManager(_sessions_dir(user.emp_no), workspace=workspace)
    except ValueError as e:
        raise HTTPException(status_code=400, detail=f"非法 workspace: {e}")
    if sm.load(parent_sid) is None:
        raise HTTPException(status_code=404, detail="父会话不存在")
    # 幂等：同父 + 同 purpose 的未归档 aux 直接复用
    for s in sm.list_sessions():
        if (s.get("kind") == "aux"
                and s.get("parent_session_id") == parent_sid
                and s.get("purpose") == purpose
                and not s.get("archived")):
            return {"ok": True, "session_id": s["id"], "created": False}
    sid = sm.create(
        emp_no=user.emp_no, workspace=workspace, purpose=purpose,
        extra_meta={"kind": "aux", "parent_session_id": parent_sid})
    try:
        from services import agent_session_store
        agent_session_store.upsert_session(sid, user.emp_no, workspace, purpose, "")
    except Exception:
        pass
    return {"ok": True, "session_id": sid, "created": True}
```

```python
# 辅助会话：父会话每轮快照注入（设计 2026-08-25 §B4，主→辅单向通道）
# ══════════════════════════════════════════════════════════

_SNAPSHOT_MAX_CHARS = 8 * 1024


def _parent_snapshot(sm: SessionManager, parent_sid: str, last_fp: str) -> tuple[str, str]:
    """构造父会话最新快照。返回 (注入文本, 新指纹)。

    - 父不可读 → 降级行（辅助会话继续不炸）
    - 指纹与上轮相同 → 一行「无变更」（省 token）
    - 内容：父会话 meta 概要 + 最新草稿足迹 + 最近 assistant 摘要（500 字符）
    """
    import hashlib
    msgs = None
    try:
        msgs = sm.load(parent_sid)
    except Exception:
        msgs = None
    if msgs is None:
        return "[parent_context] 父会话不可读（可能已删除）\n", last_fp

    parent_meta = sm._read_meta(sm.sessions_dir / f"{parent_sid}.jsonl") or {}

    # 最新草稿足迹（write_draft 收集——与前端 sessionDrafts 同语义）
    drafts: list[str] = []
    for m in msgs:
        if m.get("role") != "assistant" or not m.get("tool_calls"):
            continue
        for tc in m["tool_calls"]:
            try:
                args = json.loads(tc.get("function", {}).get("arguments", "{}"))
            except (json.JSONDecodeError, TypeError):
                continue
            fname = tc.get("function", {}).get("name", "")
            if (fname == "write_draft" or (fname == "draft" and args.get("action") == "write")) \
                    and args.get("filename"):
                drafts.append(str(args["filename"]))

    last_assistant = ""
    for m in reversed(msgs):
        if m.get("role") == "assistant" and (m.get("content") or "").strip():
            last_assistant = m["content"][:500]
            break

    parts = [
        f"父会话 {parent_sid}（title={parent_meta.get('title', '')} "
        f"purpose={parent_meta.get('purpose', '')}）最新快照：",
        f"- 最新草稿文件: {', '.join(drafts[-5:]) or '（无）'}",
        f"- 最近回复摘要: {last_assistant or '（无）'}",
    ]
    body = "\n".join(parts)
    fp = hashlib.sha256(body.encode("utf-8")).hexdigest()[:12]
    if last_fp and fp == last_fp:
        return "[parent_context] 主会话自上轮起无变更\n", fp
    if len(body) > _SNAPSHOT_MAX_CHARS:
        body = body[:_SNAPSHOT_MAX_CHARS] + "\n…（截断）"
    return f"<parent_context>\n{body}\n</parent_context>\n", fp
```

```tsx
  const artifacts = (() => {

          </div>
```
