```python
                        "name": str(u.get("name", "") or ""),
                        "scopes": scopes}
    return list(out.values())
```

```python
def sync_minutes(specflow_emp_base: Path, days: int = 30,
                 owner_ids: list[str] | None = None) -> list[dict]:
    """妙记快照落池（minutes +search → 逐个 +detail）。

    owner_ids=None → --owner-ids me（现状）；指定时按 open_id 检索（用户监听）。
    """
    if days <= 0:
        return []
    owners = ",".join(owner_ids) if owner_ids else "me"
    found = _run_lark(["minutes", "+search", "--owner-ids", owners,
                       "--page-size", "20"])
    items = (found.get("data") or {}).get("items") or []
    out = []
    for it in items:
        token = str(it.get("minute_token") or it.get("token") or "")
        if not token:
            continue
        detail = _run_lark(["minutes", "+detail", "--minute-tokens", token,
                            "--summary", "--keyword"])
        out.append(pool_store.save_minutes_snapshot(
            specflow_emp_base, token, detail.get("data") or {},
            source_url=str(it.get("url") or "")))
    return out
```

```python
def sync_user_materials(specflow_emp_base: Path) -> dict:
    """用户材料轮询（设计 §4.3）：按 users[].scopes 拉指定用户妙记/文档落池。

    单用户/单文档失败记 errors 不中断（诚实呈现）。
    """
    cfg = load_watch_config(specflow_emp_base)
    out = {"minutes": 0, "docs": 0, "errors": []}
    minutes_days = int(cfg.get("minutes_days") or 0)
    for u in cfg.get("users") or []:
        open_id = str(u.get("open_id", ""))
        scopes = set(u.get("scopes") or [])
        if not open_id:
            continue
        label = str(u.get("name") or "") or open_id
        if "minutes" in scopes:
            try:
                out["minutes"] += len(sync_minutes(
                    specflow_emp_base, minutes_days, owner_ids=[open_id]))
            except Exception as e:
                out["errors"].append(f"minutes {label}: {e}")
        if "docs" in scopes:
            r = _sync_user_docs(specflow_emp_base, open_id)
            out["docs"] += r["docs"]
            out["errors"].extend(f"docs {label}: {e}" for e in r["errors"])
    return out
```

```python
def _sync_user_docs(specflow_emp_base: Path, open_id: str,
                    query: str = "") -> dict:
    """按用户检索文档落池：docs +search（空 query 实测可用）→ 应用侧按
    result_meta.owner_id 过滤（CLI --filter owner_ids 2026-08-17 实测不生效）。

    召回边界：当前用户可见的近期文档 ∩ 该用户产出（诚实呈现，见设计 §4.5）。
    """
    out = {"docs": 0, "errors": []}
    found = _run_lark(["docs", "+search", "--as", "user",
                       "--query", query, "--page-size", "50"])
    results = (found.get("data") or {}).get("results") or []
    for it in results:
        meta = it.get("result_meta") or {}
        if str(meta.get("owner_id", "")) != open_id:
            continue
        token = str(meta.get("token", ""))
        if not token:
            continue
        try:
            sync_doc(specflow_emp_base, token, url=str(meta.get("url") or ""))
            out["docs"] += 1
        except Exception as e:
            out["errors"].append(f"{token}: {e}")
    return out
```

```python
    def _direct_write_chat(self, ev: dict):
        """IM 事件直写（T-925）：event_id 去重 → 渲染行 → append 快照。

        撤回/编辑无 CLI 事件，由每日校对重拉兜底（设计 §4.3）。
        授权校验 + 发送人过滤（设计 D4/D5）：仅授权 chat 且（配置了
        group_sender 时）命中发送人的消息才直写；校对重拉全量不受过滤影响。
        """
        from specflow.feishu_pool import pool_store
        event_id = str(ev.get("event_id") or "")
        chat_id = str(ev.get("chat_id") or "")

            return

                emp_base, chat_id, str(ev.get("sender_id") or "")):
            return
        if event_id and not pool_store.mark_event_seen(emp_base, event_id):

    def _authorized_for_direct_write(self, emp_base: Path, chat_id: str,
                                     sender_id: str) -> bool:
        """直写授权校验（配置即时读取，保存即生效，无需重启子进程——D1）。

        1. chat_id 须在 watch.chats 授权清单内（D5：兑现同步范围即授权范围；
           p2p 事件 chat_id 天然不在清单，兼容 D2 私聊不做）；
        2. G = scopes 含 group_sender 的 users open_id 集合——G 非空时

            return True
        authorized = {str(c.get("chat_id", "")) for c in cfg.get("chats") or []}
        if chat_id not in authorized:
            return False
        senders = {str(u.get("open_id", "")) for u in cfg.get("users") or []
                   if "group_sender" in (u.get("scopes") or [])}

```

```python
def search_users(query: str) -> dict:
    """通讯录用户搜索：contact +search-user --has-chatted。
    query 为空 → 空列表（不调 CLI，避免无关键词扫全量）。"""
    out = {"items": [], "errors": []}
    query = (query or "").strip()
    if not query:
        return out
    try:
        payload = _run_lark(["contact", "+search-user", "--as", "user",
                             "--query", query, "--has-chatted"])
    except Exception as e:
        out["errors"].append(str(e))
        return out
    data = payload.get("data") or {}
    for it in (data.get("users") or data.get("items") or []):
        open_id = str(it.get("open_id", ""))
        if not open_id:
            continue
        out["items"].append({
            "open_id": open_id,
            "name": str(it.get("name") or it.get("localized_name") or ""),
            "email": str(it.get("enterprise_email") or it.get("email") or ""),
            "department": str(it.get("department") or ""),
        })
    return out


def search_user_docs(open_id: str, query: str = "") -> dict:
    """按用户搜文档：docs +search → 应用侧按 owner_id 过滤

    召回边界：当前用户可见文档 ∩ 该用户产出（诚实呈现）。"""
    out = {"items": [], "errors": []}
    open_id = (open_id or "").strip()
    if not open_id:
        out["errors"].append("open_id 必填")
        return out
    try:
        payload = _run_lark(["docs", "+search", "--as", "user",
                             "--query", query or "", "--page-size", "50"])
    except Exception as e:
        out["errors"].append(str(e))
        return out
    results = (payload.get("data") or {}).get("results") or []
    for it in results:
        meta = it.get("result_meta") or {}
        if str(meta.get("owner_id", "")) != open_id:

        token = str(meta.get("token", ""))
        if not token:
            continue
        title = str(it.get("title_highlighted") or it.get("title") or "")
        title = title.replace("<em>", "").replace("</em>", "")
        out["items"].append({
            "token": token,
            "title": title,
            "url": str(meta.get("url") or ""),
            "doc_type": str(meta.get("doc_types") or ""),
            "owner_id": open_id,
            "owner_name": str(meta.get("owner_name") or ""),
        })
```

```tsx
          {/* 用户区：搜索选择 + scopes */}
          <div className="mb-3">
            <div className="text-[11px] text-gray-500 mb-1">监听用户（群消息按发送人过滤 / 用户产出妙记 / 用户产出文档）</div>
            <div className="flex gap-2 mb-1.5">
              <input
                className="flex-1 bg-surface-3 border border-border rounded px-2 py-1 text-xs text-gray-200 outline-none focus:border-info"
                value={userQuery} onChange={e => setUserQuery(e.target.value)}
                onKeyDown={e => e.key === 'Enter' && searchUsers()}
                placeholder="姓名 / 邮箱关键词" />
              <button className={btn} onClick={searchUsers} disabled={busy === 'user-search'}>搜索</button>
            </div>
            {userResults.length > 0 && (
              <div className="max-h-28 overflow-y-auto bg-surface-3 border border-border rounded p-2 space-y-1 mb-1.5">
                {userResults.map(u => (
                  <div key={u.open_id} className="flex items-center gap-2 text-xs">
                    <span className="text-gray-200">{u.name || u.open_id}</span>
                    <span className="text-[10px] text-gray-600">
                      {u.department}{u.email ? ` · ${u.email}` : ''}</span>
                    <button className={btn} onClick={() => addUser(u)}
                      disabled={watchUsers.some(w => w.open_id === u.open_id)}>添加</button>
                  </div>
                ))}
              </div>
            )}
            {watchUsers.length > 0 && (
              <div className="bg-surface-3 border border-border rounded p-2 space-y-1">
                {watchUsers.map(u => (
                  <div key={u.open_id} className="flex items-center gap-3 text-xs">
                    <span className="text-gray-200 w-20 truncate">{u.name || u.open_id}</span>
                    {([['group_sender', '群消息'], ['minutes', '妙记'], ['docs', '文档']] as const)
                      .map(([s, label]) => (
                        <label key={s} className="flex items-center gap-1 text-gray-400 cursor-pointer">
                          <input type="checkbox" checked={u.scopes.includes(s)}
                            onChange={() => toggleScope(u.open_id, s)} />{label}
                        </label>
                      ))}
                    <button className="ml-auto text-[11px] text-red-400 hover:text-red-300"
                      onClick={() => setWatchUsers(watchUsers.filter(w => w.open_id !== u.open_id))}>
                      移除</button>
                  </div>
                ))}
              </div>
            )}
          </div>

          {/* 文档区：按用户搜索 + 手填兜底 */}
          <div className="mb-2">
            <div className="text-[11px] text-gray-500 mb-1">
              按用户搜文档（召回 = 当前用户可见文档 ∩ 该用户产出）</div>
            <div className="flex gap-2 mb-1.5">
              <select className="bg-surface-3 border border-border rounded px-2 py-1 text-xs text-gray-200"
                value={docOwner} onChange={e => setDocOwner(e.target.value)}>
                <option value="">选择用户</option>
                {watchUsers.filter(u => u.scopes.includes('docs')).map(u => (
                  <option key={u.open_id} value={u.open_id}>{u.name || u.open_id}</option>
                ))}
              </select>
              <input
                className="flex-1 bg-surface-3 border border-border rounded px-2 py-1 text-xs text-gray-200 outline-none focus:border-info"
                value={docQuery} onChange={e => setDocQuery(e.target.value)}
                onKeyDown={e => e.key === 'Enter' && searchUserDocs()}
                placeholder="关键词（可空）" />
              <button className={btn} onClick={searchUserDocs} disabled={busy === 'doc-search'}>搜索</button>
            </div>
            {docSelList.length > 0 && (
              <div className="max-h-28 overflow-y-auto bg-surface-3 border border-border rounded p-2 space-y-1">
                {docSelList.map(d => (
                  <label key={d.token} className="flex items-center gap-2 text-xs cursor-pointer">
                    <input type="checkbox" checked={docSel.has(d.token)}
                      onChange={e => {
                        const next = new Set(docSel);
                        if (e.target.checked) next.add(d.token); else next.delete(d.token);
                        setDocSel(next);
                      }} />
                    <span className="text-gray-200">{d.title || d.token}</span>
                    <span className="text-[10px] text-gray-600">{d.doc_type} · {d.owner_name}</span>
                  </label>
                ))}
              </div>
            )}
            <div className="text-[11px] text-gray-500 mt-2 mb-1">手填文档 token（每行一个，与勾选合并去重）</div>
            <textarea
              className="w-full h-16 bg-surface-3 border border-border rounded px-2 py-1.5 text-xs text-gray-200 outline-none focus:border-info"
              value={watchDocs} onChange={e => setWatchDocs(e.target.value)}
              placeholder="FBxodQkaJoUicrxdk67cUkbhnwc" />
          </div>

          <div className="text-[11px] text-gray-500">
            妙记同步窗口 {watch.minutes_days} 天 · 群聊窗口 {watch.chat_days} 天（在 feishu_watch.yaml 中调整）
          </div>
        </div>
      )}
```
