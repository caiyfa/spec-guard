> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
> 设计依据：`docs/远期设计/SpecGuard-飞书知识图谱设计-面试素材溯源.md`（v1.0，已评审）

**Goal:** 飞书材料（文档/妙记/群聊话题）经「拉取落池 → LLM 建议 → 人工复核 → 图谱派生」管线接入 graph_index，面试官可沿 trace_backward 追溯素材证据链，工作台提供复核面板与双层图谱视图。

**Architecture:** 五层流——采集层（feishu_pool 私有区落盘）→ 提炼层（LLM 话题聚类 + 建边建议，产出落盘）→ 闸门层（复核确认写共享区事实源）→ 派生层（graph_index.builder 飞书段）→ 呈现层（复核面板 + 焦点子图 + 网状镜头）。铁律：LLM 不直接写图谱；私有/共享双区隔离；删库重建同图。

**Tech Stack:** Python 3.14（specflow / graph_index / FastAPI）、SQLite（workbench.db + graph_index.db）、React 19 + @xyflow/react（已装）+ elkjs、lark-cli（只读命令）、pytest。

"""paths.py — 飞书材料池路径函数（唯一事实源，飞书知识图谱一期）。

私有区：.specflow/{emp}/feishu_pool/（随用户，原始快照永不出私有区）

from __future__ import annotations
from pathlib import Path


def pool_dirs(specflow_emp_base: Path) -> dict:
    """私有区材料池目录（docs/minutes/chats + imgmap 与原文同目录）。"""
    base = Path(specflow_emp_base)
    return {
        "root": base / "feishu_pool",
        "docs": base / "feishu_pool" / "docs",
        "minutes": base / "feishu_pool" / "minutes",
        "chats": base / "feishu_pool" / "chats",
    }


def shared_dirs(workspace_root: Path) -> dict:
    """共享区事实源目录（topics.yaml / material_links.yaml / materials/）。"""
    base = Path(workspace_root) / ".specguard" / "feishu"
    return {"root": base, "materials": base / "materials",
            "topics_yaml": base / "topics.yaml",
            "links_yaml": base / "material_links.yaml"}
```

# specflow/feishu_pool/pool_store.py（骨架，关键函数完整）
"""pool_store.py — 材料快照落盘/读取（原文冻结可复算，imgmap 分离）。"""
from __future__ import annotations
import hashlib, json
from pathlib import Path
from specflow.feishu_pool.paths import pool_dirs


def sha12(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()[:12]


def chat_sha8(chat_id: str) -> str:
    """群 id 脱敏（原始 chat_id 只在私有区配置与快照路径出现前的映射里）。"""
    return hashlib.sha256(chat_id.encode("utf-8")).hexdigest()[:8]


def save_doc_snapshot(specflow_emp_base: Path, doc_token: str, markdown: str,
                      imgmap: dict | None = None, source_url: str = "") -> dict:
    d = pool_dirs(specflow_emp_base)["docs"]
    d.mkdir(parents=True, exist_ok=True)
    stem = f"fd-{doc_token}"
    (d / f"{stem}.md").write_text(markdown, encoding="utf-8")
    if imgmap is not None:
        (d / f"{stem}.imgmap.json").write_text(
            json.dumps(imgmap, ensure_ascii=False, indent=2), encoding="utf-8")
    return {"node_id": f"fd:{doc_token}", "sha": sha12(markdown),
            "source_url": source_url, "file": f"docs/{stem}.md"}


def save_minutes_snapshot(specflow_emp_base: Path, minute_token: str,
                          detail_json: dict, source_url: str = "") -> dict:
    d = pool_dirs(specflow_emp_base)["minutes"]
    d.mkdir(parents=True, exist_ok=True)
    stem = f"fm-{minute_token}"
    raw = json.dumps(detail_json, ensure_ascii=False, indent=2)
    (d / f"{stem}.json").write_text(raw, encoding="utf-8")
    return {"node_id": f"fm:{minute_token}", "sha": sha12(raw),
            "source_url": source_url, "file": f"minutes/{stem}.json"}


def save_chat_snapshot(specflow_emp_base: Path, chat_id: str, day: str,
                       markdown: str) -> dict:
    cs8 = chat_sha8(chat_id)
    d = pool_dirs(specflow_emp_base)["chats"] / cs8
    d.mkdir(parents=True, exist_ok=True)
    (d / f"{day}.md").write_text(markdown, encoding="utf-8")
    return {"chat_sha8": cs8, "day": day, "sha": sha12(markdown),
            "file": f"chats/{cs8}/{day}.md"}
```
