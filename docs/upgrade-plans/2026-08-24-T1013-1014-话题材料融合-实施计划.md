# 话题材料融合实施计划（T-1013 / T-1014）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** ①「更新本群」单群同步覆盖文档/妙记通道；②话题材料页签下线群聊行（与话题页签原文时间轴重复展示）；③建立话题↔文档/妙记的 references 关联（确定性预筛 + 人工确认 + 话题卡挂载），把文档/妙记合并进话题模块。

**Architecture:** 后端在 specflow 新增 `topic_material.py` 纯函数预筛模块（零 LLM，仿 T-993 cross_chat 两段式）；`list_pool_topics` 响应时派生每话题 `materials` 与全局 `material_suggestions`（不落库，同 T-992 modules 口径）；路由新增 `/topics/material-relate` 端点复用 T-993 relates 的「直接人工确认」链路（不进复核队列——复核队列保持专职 FR mentions）；graph_index builder **零改动**（references 边按 edge 字段分派自然承接，节点存在性守卫兜底断链）。前端 FeishuPool 材料页签只承载文档/妙记；TopicCard 新增关联材料 chips；TopicWorkspace 新增材料推荐面板。

**关键设计决策（已拍板口径，执行者勿再发散）：**
- 边方向：`src=tp:*, dst=fd:*/fm:*`，edge=`references`（语义"话题引用材料"；builder 的 relates tp↔tp 守卫不受影响）
- 关联确认走 T-993 同款直接人工确认（话题页内推荐面板），**不复用复核队列**——队列语义是 FR mentions 建边，混进去会污染「材料管理」页
- LLM 精筛本期不做（YAGNI：候选量小、预筛 reasons 透明可读；如需精筛后续仿 /topics/cross-refine 加端点）
- 群聊材料**不**进入推荐范围（话题已通过 message_ids/window 直连群聊原文，重复建边无意义）
- 图片不单独处理：随消息时间轴（chats imgmap）与文档详情（docs imgmap）既有链路可达

**Tech Stack:** Python 3.14 / FastAPI / pytest（test_specstd）；React 19 / TS 严格模式 / Vite 6 / oxlint / vitest

```python
动态求值，任何时区下稳定。

from datetime import datetime

from specflow.feishu_pool import topic_material as tm


def _topic(**over):
    t = {"id": "tp:aaaaaa-20260807-01", "title": "断档分析报告排查",
         "summary": "s", "keywords": ["断档", "邮箱"], "category": "requirement",
         "status": "draft",
         "window": ["2026-08-07T14:00", "2026-08-07T15:00"]}
    t.update(over)
    return t


def _mat(**over):
    m = {"id": "fm:abc", "type": "feishu_minutes", "title": "周会",
         "summary": "s", "keywords": ["断档", "排期"], "file": "minutes/fm-abc.json",
         "mtime": datetime(2026, 8, 7, 16, 0).timestamp()}
    m.update(over)
    return m


def test_hit_by_minutes_keyword_intersection():
    out = tm.suggest_topic_materials([_topic()], [_mat()], done=set())
    assert [(s["src"], s["dst"]) for s in out] == [("tp:aaaaaa-20260807-01", "fm:abc")]
    assert any("断档" in r for r in out[0]["reasons"])


def test_hit_by_doc_text_contains():
    """文档无结构化关键词：话题关键词在材料 title+summary 文本包含即命中。"""
    m = _mat(id="fd:tokA", type="feishu_doc", title="数据口径说明",
             keywords=[], summary="本文说明断档统计口径")
    out = tm.suggest_topic_materials([_topic()], [m], done=set())
    assert len(out) == 1 and out[0]["dst"] == "fd:tokA"


def test_hit_by_title_similarity_without_keywords():
    t = _topic(keywords=[], title="断档分析报告排查")
    m = _mat(keywords=[], title="断档分析报告排查纪要")
    out = tm.suggest_topic_materials([t], [m], done=set())
    assert len(out) == 1 and any("标题相似" in r for r in out[0]["reasons"])
```

```python
"""topic_material.py — 话题↔材料（文档/妙记）关联推荐（T-1014，仿 T-993 D5）。

确定性预筛（零 LLM，纯读）+ 人工确认写 references 边（material_links.yaml）。
只推荐不自动挂边；群聊快照不在推荐范围（话题已通过 message_ids/window
直连群聊原文，重复建边无意义）。

预筛条件（全满足才入选）：
- 话题非 rejected 且 category != chat（闲聊不关联材料）
- 材料类型限 feishu_doc / feishu_minutes
- 时间接近：材料 mtime 与话题 window 重叠或间距 ≤7 天（window 缺失放行——
  诚实不拦，靠语义条件把关，同 cross_chat._windows_close 口径）
- 语义沾边：标题相似 ≥0.4（SequenceMatcher）或关键词命中 ≥1
  （妙记带 artifacts.keywords 直接求交；文档无关键词，用话题 keywords
  在材料 title+summary 文本中的包含命中）
- (tp, mat) 未被 confirmed/rejected（references 有向边，键不归一化）
"""

from __future__ import annotations

from datetime import datetime, timedelta
from difflib import SequenceMatcher

_TITLE_SIM = 0.4
_WINDOW_DAYS = 7


def _parse_ts(v) -> datetime | None:
    s = str(v or "")
    try:
        return datetime.fromisoformat(s)
    except ValueError:
        return None


def _mtime_near_window(mtime: float, window: list) -> bool:
    """材料 mtime（秒级时间戳）与话题 window（ISO 起止）间距 ≤7 天。"""
    if not window:
        return True                     # 窗口缺失不拦（诚实放行）
    t0 = _parse_ts(window[0])
    t1 = _parse_ts(window[-1] if len(window) > 1 else None) or t0
    if t0 is None or not mtime:
        return True
    mt = datetime.fromtimestamp(float(mtime))
    return t0 - timedelta(days=_WINDOW_DAYS) <= mt \
        <= (t1 or t0) + timedelta(days=_WINDOW_DAYS)


def _keyword_hits(topic: dict, material: dict) -> list[str]:
    """关键词命中清单：妙记求交集；文档看话题关键词在材料文本中的包含。"""
    kws = [str(k) for k in (topic.get("keywords") or []) if str(k)]
    if not kws:
        return []
    m_kws = {str(k) for k in (material.get("keywords") or []) if str(k)}
    if m_kws:                           # 妙记：结构化关键词求交
        return sorted(k for k in kws if k in m_kws)
    text = f"{material.get('title') or ''}\n{material.get('summary') or ''}"
    return [k for k in kws if k in text]


def suggest_topic_materials(topics: list[dict], materials: list[dict],
                            done: set[tuple[str, str]]) -> list[dict]:
    """确定性预筛：话题 × 文档/妙记 → 候选关联（纯读零 LLM）。

    Args:
        topics: 话题卡列表（候选 + 已确认，调用方组装）
        materials: 材料清单（list_pool_materials 条目；chat 类型函数内过滤）
        done: 已处理 (tp_id, material_id) 对（confirmed/rejected 均不再推荐）

    Returns:
        [{src, dst, score, reasons}]，按 score 降序；
        score = 标题相似度 + 0.2 × 关键词命中数（上限 1.0）
    """
    cards = [t for t in (topics or [])
             if isinstance(t, dict) and t.get("id")
             and str(t.get("status") or "") != "rejected"
             and str(t.get("category") or "routine") != "chat"]
    mats = [m for m in (materials or [])
            if isinstance(m, dict)
            and m.get("type") in ("feishu_doc", "feishu_minutes") and m.get("id")]
    done = done or set()
    out: list[dict] = []
    for t in cards:
        for m in mats:
            key = (str(t["id"]), str(m["id"]))
            if key in done:
                continue
            if not _mtime_near_window(float(m.get("mtime") or 0),
                                      t.get("window") or []):
                continue
            title_sim = SequenceMatcher(
                None, str(t.get("title", "")),
                str(m.get("title", ""))).ratio()
            hits = _keyword_hits(t, m)
            if title_sim < _TITLE_SIM and not hits:
                continue
            reasons = []
            if title_sim >= _TITLE_SIM:
                reasons.append(f"标题相似 {title_sim:.2f}")
            if hits:
                reasons.append(f"关键词命中 {hits}")
            score = min(1.0, title_sim + 0.2 * len(hits))
            out.append({"src": key[0], "dst": key[1],
                        "score": round(score, 2), "reasons": reasons})
    out.sort(key=lambda c: (-c["score"], c["src"], c["dst"]))
    return out


def materials_by_topic(links: list[dict],
                       mat_index: dict[str, dict]) -> dict[str, list[dict]]:
    """confirmed references 边 → {tp_id: [{id,type,title,file}]}（话题卡挂材料区）。

    断链诚实跳过（材料已删/未同步的 dst 不出渣数据）；同话题下按材料 id 排序。
    """
    out: dict[str, list[dict]] = {}
    for l in links or []:
        if str(l.get("edge") or "") != "references" \
                or l.get("status") != "confirmed":
            continue
        src, dst = str(l.get("src", "")), str(l.get("dst", ""))
        if not src.startswith("tp:"):
            continue
        m = mat_index.get(dst)
        if m is None:
            continue
        out.setdefault(src, []).append({
            "id": m["id"], "type": m["type"],
            "title": str(m.get("title") or m["id"]),
            "file": str(m.get("file") or ""),
        })
    for v in out.values():
        v.sort(key=lambda x: x["id"])
    return out
```

```python
def _minutes_digest(f: Path) -> tuple[str, str, list[str]]:
    """妙记快照提取 title/summary/keywords（T-1014：artifacts.keywords 供
    话题↔材料预筛；minutes 数组或裸对象兼容；坏 JSON 诚实降级空值）。"""
    try:
        data = json.loads(f.read_text(encoding="utf-8", errors="replace"))
    except Exception:
        return "", "", []
    entries = data.get("minutes") if isinstance(data, dict) else None
    if not entries and isinstance(data, dict):
        entries = [data]
    for it in entries or []:
        if not isinstance(it, dict):
            continue
        arts = it.get("artifacts") or {}
        kws = [str(k) for k in (arts.get("keywords") or []) if str(k)]
        return (str(it.get("title") or ""),
                str(arts.get("summary") or it.get("summary") or ""), kws)
    return "", "", []
```

```python
# T-1014：话题↔材料关联（references 边——只推荐不自动挂，人确认后生效）
# ══════════════════════════════════════════════════════════

@router.post("/topics/material-relate")
def relate_topic_material(body: MaterialRelateRequest,
                          user: CurrentUser = Depends(get_current_user),
                          workspace: str = Depends(resolve_workspace)):
    """人工确认/忽略话题↔材料关联：写 material_links.yaml 的 references 边
    （confirmed=生效进图谱 + 材料副本提升图谱派生源 / rejected=留痕防重复
    推荐）。只挂边不改材料本体；群聊材料不在范围（话题已 message_ids 直连）。
    """
    from specflow.feishu_pool import facts
    from specflow.feishu_pool import topic_cluster as tc
    known = {str(t.get("id")) for t in tc._load_candidates(_material_base())}
    if body.topic_id not in known:
        raise HTTPException(status_code=404,
                            detail=f"话题不存在: {body.topic_id}")
    mats = {str(m["id"]) for m in sync_svc.list_pool_materials(_material_base())
            if m.get("type") in ("feishu_doc", "feishu_minutes")}
    if body.material_id not in mats:
        raise HTTPException(status_code=404,
                            detail=f"材料不存在: {body.material_id}")
    facts.append_link(_ws_root(), src=body.topic_id, dst=body.material_id,
                      edge="references",
                      confidence=1.0 if body.confirmed else 0.0,
                      evidence="", confirmed_by=str(user.emp_no or ""),
                      status="confirmed" if body.confirmed else "rejected")
    if body.confirmed:
        from services.specflow import feishu_review_service as rev_svc
        rev_svc.promote_node(_material_base(), _ws_root(), body.material_id)
    return {"src": body.topic_id, "dst": body.material_id,
            "status": "confirmed" if body.confirmed else "rejected"}
```
