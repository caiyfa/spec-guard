## 二、接口契约（B-1 冻结，前端按此开发）

```typescript
// workbench/frontend/src/types.ts

type StreamEventType =
  | "text_delta"
  | "reasoning_delta"
  | "tool_call_start"
  | "tool_call_delta"
  | "done"
  | "error";

interface StreamEvent {
  type: StreamEventType;
  content?: string;                // text_delta / reasoning_delta
  tool_call?: {                    // tool_call_start
    id: string;
    type: string;
    function: { name: string; arguments: string };
  };
  index?: number;                  // tool_call_*（多工具时按 index 区分）
  delta?: string;                  // tool_call_delta（参数增量）
  message?: {                      // done
    role: string;
    content: string | null;
    tool_calls?: Array<...>;
  };
  usage?: {                        // done
    prompt_tokens: number;
    completion_tokens: number;
    total_tokens: number;
  };
  error?: string;                  // error
}
```

**WS 事件**（后端推送，前端消费）：

```typescript
type AgentWSEvent =
  | StreamEvent                              // LLM 流式增量
  | { type: "assistant_delta"; delta: string }      // runtime 透传的文本增量
  | { type: "tool_call_start"; tool_call: {...} }   // 工具调用开始
  | { type: "tool_call"; tool_call: {...}; result: string }  // 工具结果
  | { type: "assistant"; message: {...}; usage: {...} }      // 整轮完成
  | { type: "state"; turn?: number; fused?: boolean }
  | { type: "compaction"; message_count: number }
  | { type: "error"; message: string };
```

## 三、前端改造（`workbench/frontend/src/pages/AgentChat.tsx`）

### 核心逻辑

```tsx
// 新增状态
const [streamingContent, setStreamingContent] = useState("");
const [toolCalls, setToolCalls] = useState<ToolCallState[]>([]);

useEffect(() => {
  ws.onmessage = (e) => {
    const event = JSON.parse(e.data);
    switch (event.type) {
      case "assistant_delta":
        // 逐字渲染
        setStreamingContent(prev => prev + event.delta);
        break;
      case "reasoning_delta":
        // 思考过程（可选展示，如折叠面板）
        setReasoningContent(prev => prev + event.content);
        break;
      case "tool_call_start":
        // 显示工具调用 loading 卡片
        setToolCalls(prev => [...prev, {
          id: event.tool_call.id,
          name: event.tool_call.function.name,
          status: "running",
        }]);
        break;
      case "tool_call":
        // 更新工具结果
        setToolCalls(prev => prev.map(tc =>
          tc.id === event.tool_call.id
            ? { ...tc, status: "done", result: event.result }
            : tc
        ));
        break;
      case "assistant":
        // 整轮定稿，清空 streaming
        setMessages(prev => [...prev, event.message]);
        setStreamingContent("");
        setStats(event.usage);
        break;
      case "error":
        setError(event.message);
        break;
    }
  };
}, []);
```

### 工具调用卡片组件

```tsx
function ToolCallCard({ toolCall }: { toolCall: ToolCallState }) {
  return (
    <Card>
      <Badge status={toolCall.status === "running" ? "processing" : "success"}>
        {toolCall.status === "running" ? "调用中..." : "完成"}
      </Badge>
      <Text code>{toolCall.name}</Text>
      {toolCall.status === "done" && (
        <Collapse>
          <pre>{toolCall.result}</pre>
        </Collapse>
      )}
    </Card>
  );
}
```

## 四、与 B-1 的并行开发策略

**B-1 冻结接口契约后，B-3 即可并行启动**：
- B-1 负责后端 `chat_stream` 实现（产出 `StreamEvent`）
- B-3 负责前端按契约渲染（消费 `StreamEvent`）

**并行要点**：
- 接口契约（`StreamEvent` 结构）必须先冻结（B-1 的交付物之一）
- B-3 用 **mock 数据**开发（前端模拟 `StreamEvent` 事件流），不依赖 B-1 的真实实现
- B-1 完成后，联调验证（替换 mock 为真实 WS 事件）
